  <action>
Create pwa/scripts/create-icons.js using Node.js canvas:
```javascript
const { createCanvas } = require('canvas');
const { writeFile } = require('fs/promises');
const { join } = require('path');

const sizes = [192, 512];

async function createIcons() {
  for (const size of sizes) {
    const canvas = createCanvas(size, size);
    const ctx = canvas.getContext('2d');

    // Black background
    ctx.fillStyle = '#000000';
    ctx.fillRect(0, 0, size, size);

    // White "EW" text
    ctx.fillStyle = '#FFFFFF';
    ctx.font = `bold ${size * 0.4}px Arial`;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText('EW', size / 2, size / 2);

    // Export as PNG
    const buffer = canvas.toBuffer('image/png');
    const filename = `icon-${size}x${size}.png`;
    await writeFile(join(__dirname, '..', 'public', 'icons', filename), buffer);
    console.log(`Created ${filename}`);
  }
}

createIcons().catch(console.error);
```

Create pwa/app/hooks/use-ios-detection.tsx:
```tsx
'use client';

import { useEffect, useState } from 'react';

export function useIOSDetection() {
  const [isIOS, setIsIOS] = useState(false);
  const [isStandalone, setIsStandalone] = useState(false);

  useEffect(() => {
    const ua = window.navigator.userAgent;
    const iOS = /iPad|iPhone|iPod/.test(ua) || (ua.includes('Mac') && 'ontouchend' in document);
    setIsIOS(iOS);

    // Check if already installed (standalone mode)
    const standalone = (window.navigator as any).standalone === true ||
      window.matchMedia('(display-mode: standalone)').matches;
    setIsStandalone(standalone);
  }, []);

  const showIOSInstructions = isIOS && !isStandalone;

  return {
    isIOS,
    isStandalone,
    showIOSInstructions,
  };
}
```

This detects iOS Safari and whether the app is already in standalone mode.

import { useIOSDetection } from '@/hooks/use-ios-detection';

export function IOSInstallInstructions() {

          </div>
          <div className="flex-1">
            <h3 className="font-semibold text-gray-900">Install on iOS</h3>

              Tap the Share button, then &quot;Add to Home Screen&quot;
            </p>
            <ol className="text-xs text-gray-500 mt-2 space-y-1 list-decimal list-inside">
              <li>Tap the Share button <span className="inline-block ml-1"><Share2 className="w-3 h-3 inline" /></span></li>
