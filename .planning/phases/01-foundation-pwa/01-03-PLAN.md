<task type="auto">
  <name>Task 5: Create service worker with Serwist</name>
  <files>pwa/public/sw.ts</files>
  <action>
Create pwa/public/sw.ts:
```typescript
import { defaultCache, setupBroadcast } from 'serwist';
import { cleanupOutdatedCaches, precacheAndRoute } from 'serwist/worker';
import { serwist } from 'serwist/worker';

declare global {
  interface WorkerGlobalScope {
    __SW_MANIFEST: (string | PrecacheEntry)[] | undefined;
  }
}

declare const self: ServiceWorkerGlobalScope;

serwist.precache([
  { url: '/', revision: __SW_MANIFEST?.find((e) => e.url === '/')?.revision },
  { url: '/manifest.json', revision: __SW_MANIFEST?.find((e) => e.url === '/manifest.json')?.revision },
]);

serwist.routeCache(
  ({
    url,
    request,
  }: {
    url: URL;
    request: Request;
  }) => {
    // Cache static assets
    return url.origin === self.location.origin && request.method === 'GET';
  },
  new defaultCache({
    cacheName: 'static-assets',
    cacheQueryOptions: {
      ignoreVary: true,
      ignoreSearch: true,
    },
  })
);

cleanupOutdatedCaches();

setupBroadcast({
  onMessage: async (event) => {
    console.log('Service worker received message:', event.data);
  },
});
```

Note: This is a simplified setup. We'll refine caching strategies in Phase 5.

```tsx
'use client';

import { useEffect } from 'react';

export function useServiceWorker() {
  useEffect(() => {
    if (
      typeof window !== 'undefined' &&
      'serviceWorker' in navigator &&
      window.workbox !== undefined
    ) {
      const wb = window.workbox;

      // Show install prompt on event
      wb.addEventListener('controlling', (event: any) => {
        if (event.isUpdate) {
          window.location.reload();
        }
      });

      wb.addEventListener('waiting', (event: any) => {
        // Notify user of update
        console.log('New service worker available, refresh to update');
      });
    }
  }, []);
}

declare global {
  interface Window {
    workbox: any;
  }
}
```

This hook will be used in Plan 01-04 for the install prompt functionality.
