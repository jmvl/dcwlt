```typescript
// pwa/app/merchant/scan-customer/page.tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import { Html5Qrcode } from 'html5-qrcode';
import { useMutation } from 'convex/react';
import { api } from '@/convex/_generated/api';
import { useRouter } from 'next/navigation';
import { useMerchantAuth } from '@/app/components/MerchantAuthProvider';
import { ScannerOverlay } from '@/app/components/ScannerOverlay';
import { X } from 'lucide-react';

export default function ScanCustomerPage() {
  const router = useRouter();
  const { merchant } = useMerchantAuth();
  const scannerRef = useRef<Html5Qrcode | null>(null);
  const [scanning, setScanning] = useState(true);
  const [result, setResult] = useState<'success' | 'error' | null>(null);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [cartAmount, setCartAmount] = useState(0);

  const processPayment = useMutation(api.cryptographicQr.processQRPayment);

  // Get cart total from merchant context
  // TODO: Wire this to actual cart context from merchant dashboard
  // For now, check if cart has items; if not, redirect back
  useEffect(() => {
    // Check for cart items - assumed from merchant cart context
    // This is a placeholder that should be replaced with actual cart integration
    const storedCart = localStorage.getItem('merchantCart');
    if (storedCart) {
      try {
        const cart = JSON.parse(storedCart);
        const total = cart.items?.reduce((sum: number, item: any) => sum + (item.price * item.quantity), 0) || 0;
        setCartAmount(total);
        if (total === 0) {
          // No items in cart - redirect back
          router.push('/merchant');
        }
      } catch {
        router.push('/merchant');
      }
    } else {
      // No cart - redirect back
      router.push('/merchant');
    }
  }, [router]);

  useEffect(() => {
    if (!scanning) return;

    const html5QrCode = new Html5Qrcode('qr-reader');
    scannerRef.current = html5QrCode;

    html5QrCode.start(
      { facingMode: 'environment' },
      { fps: 10, qrbox: { width: 250, height: 250 } },
      async (decodedText) => {
        // Parse QR URL
        const parsed = parseCustomerQR(decodedText);
        if (!parsed) {
          setErrorMessage('Invalid QR code format');
          setResult('error');
          return;
        }

        try {
          await processPayment({
            customerPrivyId: parsed.userId,
            timeKey: parsed.timeKey,
            signature: parsed.signature,
            merchantId: merchant?._id,
            amount: cartAmount,
          });

          setResult('success');
          // Vibrate on success
          navigator.vibrate?.([100, 50, 100]);

          // Clear cart after successful payment
          localStorage.removeItem('merchantCart');

          // Auto-dismiss after 3 seconds
          setTimeout(() => router.push('/merchant'), 3000);
        } catch (err: any) {
          setErrorMessage(err.message || 'Payment failed');
          setResult('error');
        }

        html5QrCode.stop();
        setScanning(false);
      },
      () => {} // Ignore scan errors
    );

    return () => {
      html5QrCode.stop().catch(() => {});
    };
  }, [scanning, merchant, cartAmount, processPayment, router]);

  // Render based on state
  if (result === 'success') {
    return (
      <div className="fixed inset-0 bg-green-500 flex items-center justify-center z-50">
        <div className="bg-white rounded-lg p-8 text-center">
          <h2 className="text-2xl font-bold text-green-600">Payment Successful</h2>
          <p className="text-gray-600 mt-2">Amount: {cartAmount} EVT</p>
          <p className="text-sm text-gray-400">Redirecting to dashboard...</p>
        </div>
      </div>
    );
  }

  if (result === 'error') {
    return (
      <div className="fixed inset-0 bg-red-500 flex items-center justify-center z-50">
        <div className="bg-white rounded-lg p-8 text-center">
          <h2 className="text-2xl font-bold text-red-600">Payment Failed</h2>
          <p className="text-gray-600 mt-2">{errorMessage}</p>
          <button
            onClick={() => {
              setResult(null);
              setScanning(true);
            }}
            className="mt-4 px-4 py-2 bg-red-500 text-white rounded"
          >
            Try Again
          </button>
        </div>
      </div>
    );
  }

  return (
    <div className="fixed inset-0 bg-black z-50">
      {/* Camera view */}
      <div id="qr-reader" className="w-full h-full">
        <ScannerOverlay>
          <div className="text-white text-center absolute bottom-32 left-0 right-0">
            <p className="text-lg">Scan customer QR code</p>
            <p className="text-sm opacity-70">Point camera at customer&apos;s phone</p>
            <p className="text-sm font-bold mt-2">Cart total: {cartAmount} EVT</p>
          </div>
        </ScannerOverlay>
      </div>

      {/* Close button */}
      <button
        onClick={() => router.push('/merchant')}
        className="absolute top-4 right-4 p-2 rounded-full bg-white/10 hover:bg-white/20"
      >
        <X className="w-6 h-6 text-white" />
      </button>
    </div>
  );
}

function parseCustomerQR(url: string): { userId: string; timeKey: string; signature: string } | null {
  try {
    const parsed = new URL(url);
    if (parsed.protocol !== 'dcwlt:') return null;
    if (parsed.pathname !== '/pay') return null;

    return {
      userId: parsed.searchParams.get('u') || '',
      timeKey: parsed.searchParams.get('k') || '',
      signature: parsed.searchParams.get('s') || '',
    };
  } catch {
    return null;
  }
}
```
