// pwa/convex/cryptographicQr.ts
import { mutation, internalQuery } from "./_generated/server";
import { v } from "convex/values";
import { internal } from "./_generated/api";

// Server secret from environment (never exposed to client)
const QR_SIGNING_SECRET = process.env.QR_SIGNING_SECRET || "default-dev-secret";

// Helper: Derive per-user secret
function deriveUserSecret(userId: string): string {
  const crypto = require("crypto");
  return crypto
    .createHmac("sha256", QR_SIGNING_SECRET)
    .update(userId)
    .digest("hex");
}

// Helper: Verify TOTP time key (5-minute window, 2-minute skew)
function verifyTimeKey(timeKey: string, derivedSecret: string): boolean {
  const otplib = require("otplib");
  const authenticator = new otplib.authenticator.Authenticator();
  authenticator.options = {
    step: 300, // 5 minutes
    window: 2, // ±2 minutes clock skew
  };
  return authenticator.check(timeKey, derivedSecret);
}

// Helper: Verify HMAC signature
function verifyHMAC(message: string, signature: string, secret: string): boolean {
  const crypto = require("crypto");
  const expectedSignature = crypto
    .createHmac("sha256", secret)
    .update(message)
    .digest("hex");
  return signature === expectedSignature;
}

// Main payment mutation
export const processQRPayment = mutation({
  args: {
    customerPrivyId: v.string(),
    timeKey: v.string(),
    signature: v.string(),
    merchantId: v.id("merchants"),
    amount: v.number(),
  },
  handler: async (ctx, args) => {
    // 1. Get derived secret for this user
    const derivedSecret = deriveUserSecret(args.customerPrivyId);

    // 2. Verify signature
    const message = `${args.customerPrivyId}:${args.timeKey}`;
    if (!verifyHMAC(message, args.signature, derivedSecret)) {
      throw new Error("Invalid QR signature");
    }

    // 3. Verify time key (5-minute validity, 2-minute skew)
    if (!verifyTimeKey(args.timeKey, derivedSecret)) {
      throw new Error("QR code expired");
    }

    // 4. Get customer wallet
    const customer = await ctx.db
      .query("users")
      .withIndex("by_privy_id", (q) => q.eq("privyId", args.customerPrivyId))
      .first();
    if (!customer) {
      throw new Error("Customer not found");
    }

    const customerWallet = await ctx.db
      .query("wallets")
      .withIndex("by_user", (q) => q.eq("userId", customer._id))
      .first();
    if (!customerWallet || customerWallet.balance < args.amount) {
      throw new Error("Insufficient balance");
    }

    // 5. Get merchant wallet
    const merchant = await ctx.db.get(args.merchantId);
    if (!merchant) {
      throw new Error("Merchant not found");
    }

    const merchantWallet = await ctx.db
      .query("wallets")
      .withIndex("by_user", (q) => q.eq("userId", merchant.userId))
      .first();
    if (!merchantWallet) {
      throw new Error("Merchant wallet not found");
    }

    // 6. Atomic balance transfer
    await ctx.db.patch(customerWallet._id, {
      balance: customerWallet.balance - args.amount,
    });
    await ctx.db.patch(merchantWallet._id, {
      balance: merchantWallet.balance + args.amount,
    });

    // 7. Create transaction record
    const transactionId = await ctx.db.insert("transactions", {
      fromWalletAddress: customerWallet.walletAddress,
      toWalletAddress: merchantWallet.walletAddress,
      amount: args.amount,
      type: "payment",
      status: "completed",
      createdAt: Date.now(),
      merchantId: args.merchantId,
    });

    return {
      success: true,
      amount: args.amount,
      transactionId,
    };
  },
});
